import React, { useEffect, useState } from 'react';
import { 
  Card, 
  CardContent, 
  Typography, 
  Box, 
  CircularProgress, 
  Divider,
  Paper,
  Chip,
  Rating,
  LinearProgress,
  Grid,
  Avatar
} from '@mui/material';
import axios from 'axios';
import FacebookIcon from '@mui/icons-material/Facebook';
import ThumbUpIcon from '@mui/icons-material/ThumbUp';
import ThumbDownIcon from '@mui/icons-material/ThumbDown';
import SentimentSatisfiedAltIcon from '@mui/icons-material/SentimentSatisfiedAlt';
import SentimentVeryDissatisfiedIcon from '@mui/icons-material/SentimentVeryDissatisfied';
import StarIcon from '@mui/icons-material/Star';
import MessageIcon from '@mui/icons-material/Message';
import PeopleIcon from '@mui/icons-material/People';

// Color constants to match your theme
const COLOR = {
  primary: "#FFCC80",
  secondary: '#e6c0c7',
  accent: '#d3a9b0',
  light: '#f8eaed',
  border: 'rgba(241, 206, 212, 0.5)',
  text: '#806368',
  facebook: {
    primary: '#1877F2',
    light: '#e9f0fe',
    border: 'rgba(24, 119, 242, 0.3)'
  }
};

const FacebookReviewInsights = ({ businessId }) => {
    const [insights, setInsights] = useState(null);
    const [loading, setLoading] = useState(false);
    // Mock data for engagement metrics - in a real app, this would come from your API
    const [engagementMetrics, setEngagementMetrics] = useState({
        likes: 245,
        comments: 78,
        shares: 32,
        recentEngagement: 18 // percentage increase in last 30 days
    });
    const [overallRating, setOverallRating] = useState(4.2);
    
    // Rating distribution for Facebook (mock data)
    const [ratingsDistribution, setRatingsDistribution] = useState({
        5: 60,
        4: 22,
        3: 12,
        2: 4,
        1: 2
    });

    const fetchInsights = async () => {
        try {
            setLoading(true);
            // Use the Facebook insights endpoint
            const res = await axios.get(`http://52.3.145.159:8080/api/v1/facebook-insights/${businessId}`);
            
            // Generate mock data for visual elements based on insights
            if (res.data) {
                // Analyze sentiment in the insights to determine a mock overall rating
                const positiveText = res.data.positivePoints.toLowerCase();
                const negativeText = res.data.negativePoints.toLowerCase();
                
                // Very basic sentiment analysis - just for demo purposes
                const positiveWords = ["appreciate", "satisfied", "thorough", "efficient", "reasonable", 
                                      "clean", "express", "willingness", "clean-up", "efficient", 
                                      "praised", "thorough", "reasonable"];
                const negativeWords = ["bad", "poor", "issue", "problem", "dissatisfied", "disappointed"];
                
                let positiveCount = positiveWords.filter(word => positiveText.includes(word)).length;
                let negativeCount = negativeWords.filter(word => negativeText.includes(word)).length;
                
                // Adjust mock rating based on sentiment analysis
                let mockRating = 3.5; // default
                if (positiveCount > negativeCount) {
                    mockRating = 4 + (Math.min(positiveCount, 5) / 10);
                } else if (negativeCount > positiveCount) {
                    mockRating = 3 - (Math.min(negativeCount, 5) / 10);
                }
                
                setOverallRating(parseFloat(mockRating.toFixed(1)));
                
                // Generate mock distribution based on the overall rating
                const mockDistribution = {
                    5: Math.round(mockRating > 4 ? 45 + (mockRating - 4) * 20 : 25),
                    4: Math.round(mockRating > 4 ? 35 : mockRating > 3 ? 45 : 20),
                    3: Math.round(mockRating > 4 ? 12 : mockRating > 3 ? 18 : 35),
                    2: Math.round(mockRating > 4 ? 5 : mockRating > 3 ? 8 : 12),
                    1: Math.round(mockRating > 4 ? 3 : mockRating > 3 ? 4 : 8)
                };
                
                setRatingsDistribution(mockDistribution);
                
                // Generate mock engagement metrics based on the sentiment
                const baseEngagement = 100;
                let engagementMultiplier = 1.0;
                
                if (positiveCount > negativeCount) {
                    engagementMultiplier = 1.0 + (positiveCount * 0.2);
                } else if (negativeCount > positiveCount) {
                    engagementMultiplier = 1.0 - (negativeCount * 0.1);
                }
                
                const mockEngagement = {
                    likes: Math.round(baseEngagement * engagementMultiplier * 2.5),
                    comments: Math.round(baseEngagement * engagementMultiplier * 0.8),
                    shares: Math.round(baseEngagement * engagementMultiplier * 0.3),
                    recentEngagement: Math.round(10 + (positiveCount - negativeCount) * 2)
                };
                
                setEngagementMetrics(mockEngagement);
            }
            
            setInsights(res.data);
        } catch (err) {
            console.error("Error fetching Facebook review insights:", err);
            setInsights(null);
        } finally {
            setLoading(false);
        }
    };

    useEffect(() => {
        if (businessId) {
            fetchInsights();
        }
    }, [businessId]);

    if (loading) {
        return (
            <Box sx={{ display: 'flex', justifyContent: 'center', py: 4 }}>
                <CircularProgress sx={{ color: COLOR.facebook.primary }} />
            </Box>
        );
    }

    if (!insights) {
        return (
            <Card sx={{ mt: 4, borderRadius: 2, border: `1px solid ${COLOR.facebook.border}`, boxShadow: '0 4px 12px rgba(0,0,0,0.1)', height: '100%' }}>
                <CardContent sx={{ height: '100%', display: 'flex', justifyContent: 'center', alignItems: 'center' }}>
                    <Typography sx={{ textAlign: 'center', py: 4, color: COLOR.text }}>
                        No Facebook review insights available for this business.
                    </Typography>
                </CardContent>
            </Card>
        );
    }

    // Extract key phrases from positive and negative points
    const extractKeyPhrases = (text) => {
        // For a real app, you'd use NLP to extract true key phrases
        // This is a simple implementation for demo purposes
        const sentences = text.split('.');
        return sentences
            .filter(s => s.trim().length > 10 && s.trim().length < 100)
            .map(s => s.trim())
            .slice(0, 3);
    };

    const positiveKeyPhrases = extractKeyPhrases(insights.positivePoints);
    const negativeKeyPhrases = extractKeyPhrases(insights.negativePoints);

    return (
        <Card sx={{ 
            mt: 4, 
            borderRadius: 2, 
            border: `1px solid ${COLOR.facebook.border}`, 
            boxShadow: '0 4px 12px rgba(0,0,0,0.1)',
            height: '100%',
            display: 'flex',
            flexDirection: 'column'
        }}>
            <CardContent sx={{ flexGrow: 1, display: 'flex', flexDirection: 'column' }}>
                <Box sx={{ 
                    display: 'flex', 
                    alignItems: 'center', 
                    gap: 1.5, 
                    mb: 3,
                    pb: 2,
                    borderBottom: `1px solid ${COLOR.facebook.border}`
                }}>
                    <Avatar sx={{ bgcolor: COLOR.facebook.primary }}>
                        <FacebookIcon sx={{ color: 'white' }} />
                    </Avatar>
                    <Typography variant="h5" sx={{ fontWeight: 'bold', color: COLOR.facebook.primary }}>
                        Facebook Review Insights
                    </Typography>
                </Box>

                <Grid container spacing={3}>
                    {/* Left Column: Ratings and Engagement Overview */}
                    <Grid item xs={12} md={4}>
                        <Box sx={{ 
                            display: 'flex',
                            flexDirection: 'column',
                            gap: 2,
                            height: '100%'
                        }}>
                            {/* Rating Overview */}
                            <Paper elevation={0} sx={{ 
                                p: 2, 
                                backgroundColor: COLOR.facebook.light,
                                borderRadius: 2,
                                flexGrow: 1,
                                display: 'flex',
                                flexDirection: 'column',
                                justifyContent: 'center',
                                alignItems: 'center'
                            }}>
                                <Typography variant="h3" sx={{ fontWeight: 'bold', color: COLOR.facebook.primary, mb: 1 }}>
                                    {overallRating}
                                </Typography>
                                <Rating 
                                    value={overallRating} 
                                    precision={0.1} 
                                    readOnly 
                                    sx={{ mb: 2 }}
                                    emptyIcon={<StarIcon style={{ opacity: 0.55 }} fontSize="inherit" />}
                                />
                                <Typography variant="body2" sx={{ color: 'text.secondary', mb: 2 }}>
                                    Based on customer recommendations
                                </Typography>
                                
                                {/* Ratings distribution */}
                                <Box sx={{ width: '100%', mt: 1 }}>
                                    {[5, 4, 3, 2, 1].map((star) => (
                                        <Box key={star} sx={{ display: 'flex', alignItems: 'center', mb: 0.5 }}>
                                            <Typography sx={{ minWidth: '30px', color: 'text.secondary' }}>
                                                {star}
                                            </Typography>
                                            <LinearProgress 
                                                variant="determinate" 
                                                value={ratingsDistribution[star]} 
                                                sx={{ 
                                                    mx: 1, 
                                                    flexGrow: 1, 
                                                    height: 8, 
                                                    borderRadius: 1,
                                                    backgroundColor: 'rgba(0,0,0,0.1)',
                                                    '& .MuiLinearProgress-bar': {
                                                        backgroundColor: COLOR.facebook.primary
                                                    }
                                                }} 
                                            />
                                            <Typography sx={{ minWidth: '35px', color: 'text.secondary' }}>
                                                {ratingsDistribution[star]}%
                                            </Typography>
                                        </Box>
                                    ))}
                                </Box>
                            </Paper>
                            
                            {/* Engagement Metrics */}
                            <Paper elevation={0} sx={{ 
                                p: 2, 
                                backgroundColor: '#ffffff',
                                borderRadius: 2,
                                border: `1px solid ${COLOR.facebook.border}`
                            }}>
                                <Typography variant="subtitle1" sx={{ fontWeight: 'bold', color: COLOR.facebook.primary, mb: 2 }}>
                                    Engagement Metrics
                                </Typography>
                                
                                <Grid container spacing={2}>
                                    <Grid item xs={4}>
                                        <Box sx={{ textAlign: 'center' }}>
                                            <ThumbUpIcon sx={{ color: COLOR.facebook.primary, mb: 0.5 }} />
                                            <Typography variant="h6" sx={{ fontWeight: 'bold' }}>
                                                {engagementMetrics.likes}
                                            </Typography>
                                            <Typography variant="body2" sx={{ color: 'text.secondary' }}>
                                                Likes
                                            </Typography>
                                        </Box>
                                    </Grid>
                                    <Grid item xs={4}>
                                        <Box sx={{ textAlign: 'center' }}>
                                            <MessageIcon sx={{ color: COLOR.facebook.primary, mb: 0.5 }} />
                                            <Typography variant="h6" sx={{ fontWeight: 'bold' }}>
                                                {engagementMetrics.comments}
                                            </Typography>
                                            <Typography variant="body2" sx={{ color: 'text.secondary' }}>
                                                Comments
                                            </Typography>
                                        </Box>
                                    </Grid>
                                    <Grid item xs={4}>
                                        <Box sx={{ textAlign: 'center' }}>
                                            <PeopleIcon sx={{ color: COLOR.facebook.primary, mb: 0.5 }} />
                                            <Typography variant="h6" sx={{ fontWeight: 'bold' }}>
                                                {engagementMetrics.shares}
                                            </Typography>
                                            <Typography variant="body2" sx={{ color: 'text.secondary' }}>
                                                Shares
                                            </Typography>
                                        </Box>
                                    </Grid>
                                </Grid>
                                
                                <Box sx={{ 
                                    mt: 2, 
                                    p: 1, 
                                    backgroundColor: 'rgba(24, 119, 242, 0.08)', 
                                    borderRadius: 1, 
                                    textAlign: 'center'
                                }}>
                                    <Typography variant="body2">
                                        <span style={{ fontWeight: 'bold', color: COLOR.facebook.primary }}>
                                            {engagementMetrics.recentEngagement}% 
                                        </span> engagement growth in last 30 days
                                    </Typography>
                                </Box>
                            </Paper>
                        </Box>
                    </Grid>

                    {/* Right Column: Insights */}
                    <Grid item xs={12} md={8}>
                        <Box sx={{ height: '100%', display: 'flex', flexDirection: 'column' }}>
                            {/* Positive Points */}
                            <Paper elevation={0} sx={{ 
                                p: 2, 
                                mb: 2, 
                                backgroundColor: 'rgba(76, 175, 80, 0.08)', 
                                borderRadius: 2,
                                border: '1px solid rgba(76, 175, 80, 0.3)'
                            }}>
                                <Box sx={{ display: 'flex', alignItems: 'center', mb: 1.5 }}>
                                    <ThumbUpIcon sx={{ color: 'success.main', mr: 1 }} />
                                    <Typography variant="h6" sx={{ fontWeight: 'bold', color: 'success.dark' }}>
                                        Positive Highlights
                                    </Typography>
                                </Box>
                                
                                <Typography sx={{ mb: 1.5, color: 'text.secondary' }}>
                                    {insights.positivePoints}
                                </Typography>
                                
                                <Box sx={{ display: 'flex', flexWrap: 'wrap', gap: 1, mt: 2 }}>
                                    {positiveKeyPhrases.map((phrase, index) => (
                                        <Chip 
                                            key={index}
                                            icon={<SentimentSatisfiedAltIcon />} 
                                            label={phrase}
                                            size="small"
                                            sx={{ 
                                                backgroundColor: 'rgba(76, 175, 80, 0.15)', 
                                                borderColor: 'rgba(76, 175, 80, 0.3)',
                                                '& .MuiChip-icon': { color: 'success.main' }
                                            }} 
                                        />
                                    ))}
                                </Box>
                            </Paper>
                            
                            {/* Negative Points */}
                            <Paper elevation={0} sx={{ 
                                p: 2, 
                                mb: 2,
                                backgroundColor: 'rgba(244, 67, 54, 0.08)', 
                                borderRadius: 2,
                                border: '1px solid rgba(244, 67, 54, 0.3)'
                            }}>
                                <Box sx={{ display: 'flex', alignItems: 'center', mb: 1.5 }}>
                                    <ThumbDownIcon sx={{ color: 'error.main', mr: 1 }} />
                                    <Typography variant="h6" sx={{ fontWeight: 'bold', color: 'error.dark' }}>
                                        Areas for Improvement
                                    </Typography>
                                </Box>
                                
                                <Typography sx={{ mb: 1.5, color: 'text.secondary' }}>
                                    {insights.negativePoints}
                                </Typography>
                                
                                <Box sx={{ display: 'flex', flexWrap: 'wrap', gap: 1, mt: 2 }}>
                                    {negativeKeyPhrases.map((phrase, index) => (
                                        <Chip 
                                            key={index}
                                            icon={<SentimentVeryDissatisfiedIcon />} 
                                            label={phrase}
                                            size="small"
                                            sx={{ 
                                                backgroundColor: 'rgba(244, 67, 54, 0.15)', 
                                                borderColor: 'rgba(244, 67, 54, 0.3)',
                                                '& .MuiChip-icon': { color: 'error.main' }
                                            }} 
                                        />
                                    ))}
                                </Box>
                            </Paper>
                            
                            {/* Overall Summary */}
                            <Paper elevation={0} sx={{ 
                                p: 2,
                                borderRadius: 2,
                                border: `1px solid ${COLOR.facebook.border}`,
                                backgroundColor: COLOR.facebook.light,
                                flexGrow: 1,
                                display: 'flex',
                                flexDirection: 'column',
                                justifyContent: 'center'
                            }}>
                                <Typography variant="h6" sx={{ fontWeight: 'bold', color: COLOR.facebook.primary, mb: 1.5 }}>
                                    Summary
                                </Typography>
                                <Typography variant="body1" sx={{ fontStyle: 'italic', color: 'text.primary' }}>
                                    "{insights.insights}"
                                </Typography>
                            </Paper>
                        </Box>
                    </Grid>
                </Grid>
            </CardContent>
        </Card>
    );
};

export default FacebookReviewInsights;
